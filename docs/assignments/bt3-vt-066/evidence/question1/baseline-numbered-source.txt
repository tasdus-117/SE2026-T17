24:             members.Add(id, new LobbyMember { ConnectionId = id, SlotIndex = slot, IsHost = isHost });

26:         }
27:         public bool Remove(int id) => members.Remove(id);
28:         public bool SetReady(int id, bool ready)
29:         {

32:             members[id] = member;
33:             return true;
34:         }

468:             // have the onlineScene feature, it will be asynchronous!
469: 
470:             // setup server first
471:             SetupServer();
472: 
473:             // scene change needed? then change scene and spawn afterwards.
474:             // => BEFORE host client connects. if client auth succeeds then the
475:             //    server tells it to load 'onlineScene'. we can't do that if
476:             //    server is still in 'offlineScene'. so load on server first.
477:             if (IsServerOnlineSceneChangeNeeded())
478:             {
479:                 // call FinishStartHost after changing scene.
480:                 finishStartHostPending = true;
481:                 ServerChangeScene(onlineScene);
482:             }
483:             // otherwise call FinishStartHost directly
484:             else
485:             {
486:                 FinishStartHost();
487:             }
488:         }

490:         // This may be set true in StartHost and is evaluated in FinishStartHost
491:         bool finishStartHostPending;
492: 
493:         // FinishStartHost is guaranteed to be called after the host server was
494:         // fully started and all the asynchronous StartHost magic is finished
495:         // (= scene loading), or immediately if there was no asynchronous magic.
496:         //
497:         // note: we don't really need FinishStartClient/FinishStartServer. the
498:         //       host version is enough.
499:         void FinishStartHost()
500:         {
501:             // ConnectHost needs to be called BEFORE SpawnObjects:
502:             // https://github.com/vis2k/Mirror/pull/1249/
503:             // -> this sets NetworkServer.localConnection.
504:             // -> localConnection needs to be set before SpawnObjects because:
505:             //    -> SpawnObjects calls OnStartServer in all NetworkBehaviours
506:             //       -> OnStartServer might spawn an object and set [SyncVar(hook="OnColorChanged")] object.color = green;
507:             //          -> this calls SyncVar.set (generated by Weaver), which has
508:             //             a custom case for host mode (because host mode doesn't
509:             //             get OnDeserialize calls, where SyncVar hooks are usually
510:             //             called):
511:             //
512:             //               if (!SyncVarEqual(value, ref color))
513:             //               {
514:             //                   if (NetworkServer.localClientActive && !getSyncVarHookGuard(1uL))
515:             //                   {
516:             //                       setSyncVarHookGuard(1uL, value: true);
517:             //                       OnColorChangedHook(value);
518:             //                       setSyncVarHookGuard(1uL, value: false);
519:             //                   }
520:             //                   SetSyncVar(value, ref color, 1uL);
521:             //               }
522:             //
523:             //          -> localClientActive needs to be true, otherwise the hook
524:             //             isn't called in host mode!
525:             //
526:             // TODO call this after spawnobjects and worry about the syncvar hook fix later?
527:             NetworkClient.ConnectHost();
528: 
529:             // invoke user callbacks AFTER ConnectHost has set .activeHost.
530:             // this way initialization can properly handle host mode.
531:             //
532:             // fixes: https://github.com/MirrorNetworking/Mirror/issues/3302
533:             // where [SyncVar] hooks wouldn't be called for objects spawned in
534:             // NetworkManager.OnStartServer, because .activeHost was still false.
535:             //
536:             // TODO is there a risk of someone connecting between Listen() and FinishStartHost()?
537:             OnStartServer();
538: 
539:             // call OnStartHost AFTER SetupServer. this way we can use
540:             // NetworkServer.Spawn etc. in there too. just like OnStartServer
541:             // is called after the server is actually properly started.
542:             OnStartHost();
543: 
544:             // server scene was loaded. now spawn all the objects
545:             NetworkServer.SpawnObjects();
546: 
547:             // connect client and call OnStartClient AFTER server scene was
548:             // loaded and all objects were spawned.
549:             // DO NOT do this earlier. it would cause race conditions where a
550:             // client will do things before the server is even fully started.
551:             //Debug.Log("StartHostClient called");
552:             SetupClient();
553:             RegisterClientMessages();
554: 
555:             // InvokeOnConnected needs to be called AFTER RegisterClientMessages
556:             // (https://github.com/vis2k/Mirror/pull/1249/)
557:             HostMode.InvokeOnConnected();
558: 
559:             OnStartClient();
560:         }

571:         public void StopServer()
572:         {

574:             if (!NetworkServer.active)
575:                 return;

577:             if (authenticator != null)

676:         {
677:             if (singleton != null && singleton == this)
678:                 return true;

681:             {

707:             // so only if we didn't destroy ourselves.

790:         /// <summary>Change the server scene and all client's scenes across the network.</summary>
791:         // Called automatically if onlineScene or offlineScene are set, but it
792:         // can be called from user code to switch scenes again while the game is
793:         // in progress. This automatically sets clients to be not-ready during
794:         // the change and ready again to participate in the new scene.
795:         public virtual void ServerChangeScene(string newSceneName)
796:         {
797:             if (string.IsNullOrWhiteSpace(newSceneName))
798:             {
799:                 Debug.LogError("ServerChangeScene empty scene name");
800:                 return;
801:             }
802: 
803:             if (NetworkServer.isLoadingScene && newSceneName == networkSceneName)
804:             {
805:                 Debug.LogError($"Scene change is already in progress for {newSceneName}");
806:                 return;
807:             }
808: 
809:             // Throw error if called from client
810:             // Allow changing scene while stopping the server
811:             if (!NetworkServer.active && newSceneName != offlineScene)
812:             {
813:                 Debug.LogError("ServerChangeScene can only be called on an active server.");
814:                 return;
815:             }
816: 
817:             // Debug.Log($"ServerChangeScene {newSceneName}");
818:             NetworkServer.SetAllClientsNotReady();
819:             networkSceneName = newSceneName;
820: 
821:             // Let server prepare for scene change
822:             OnServerChangeScene(newSceneName);
823: 
824:             // set server flag to stop processing messages while changing scenes
825:             // it will be re-enabled in FinishLoadScene.
826:             NetworkServer.isLoadingScene = true;
827: 
828:             loadingSceneAsync = SceneManager.LoadSceneAsync(newSceneName);
829: 
830:             // ServerChangeScene can be called when stopping the server
831:             // when this happens the server is not active so does not need to tell clients about the change
832:             if (NetworkServer.active)
833:             {
834:                 // notify all clients about the new scene
835:                 NetworkServer.SendToAll(new SceneMessage
836:                 {
837:                     sceneName = newSceneName
838:                 });
839:             }
840: 
841:             startPositionIndex = 0;
842:             startPositions.Clear();
843:         }

845:         // This is only set in ClientChangeScene below...never on server.
846:         // We need to check this in OnClientSceneChanged called from FinishLoadSceneClientOnly
847:         // to prevent AddPlayer message after loading/unloading additive scenes
848:         SceneOperation clientSceneOperation = SceneOperation.Normal;
849: 
850:         internal void ClientChangeScene(string newSceneName, SceneOperation sceneOperation = SceneOperation.Normal, bool customHandling = false)
851:         {
852:             if (string.IsNullOrWhiteSpace(newSceneName))
853:             {
854:                 Debug.LogError("ClientChangeScene empty scene name");
855:                 return;
856:             }
857: 
858:             //Debug.Log($"ClientChangeScene newSceneName: {newSceneName} networkSceneName{networkSceneName}");
859: 
860:             // Let client prepare for scene change
861:             OnClientChangeScene(newSceneName, sceneOperation, customHandling);
862: 
863:             // After calling OnClientChangeScene, exit if server since server is already doing
864:             // the actual scene change, and we don't need to do it for the host client
865:             if (NetworkServer.active)
866:                 return;
867: 
868:             // set client flag to stop processing messages while loading scenes.
869:             // otherwise we would process messages and then lose all the state
870:             // as soon as the load is finishing, causing all kinds of bugs
871:             // because of missing state.
872:             // (client may be null after StopClient etc.)
873:             // Debug.Log("ClientChangeScene: pausing handlers while scene is loading to avoid data loss after scene was loaded.");
874:             NetworkClient.isLoadingScene = true;
875: 
876:             // Cache sceneOperation so we know what was requested by the
877:             // Scene message in OnClientChangeScene and OnClientSceneChanged
878:             clientSceneOperation = sceneOperation;
879: 
880:             // scene handling will happen in overrides of OnClientChangeScene and/or OnClientSceneChanged
881:             // Do not call FinishLoadScene here. Custom handler will assign loadingSceneAsync and we need
882:             // to wait for that to finish. UpdateScene already checks for that to be not null and isDone.
883:             if (customHandling)
884:                 return;
885: 
886:             switch (sceneOperation)
887:             {
888:                 case SceneOperation.Normal:
889:                     loadingSceneAsync = SceneManager.LoadSceneAsync(newSceneName);
890:                     break;
891:                 case SceneOperation.LoadAdditive:
892:                     // Ensure additive scene is not already loaded on client by name or path
893:                     // since we don't know which was passed in the Scene message
894:                     if (!SceneManager.GetSceneByName(newSceneName).IsValid() && !SceneManager.GetSceneByPath(newSceneName).IsValid())
895:                         loadingSceneAsync = SceneManager.LoadSceneAsync(newSceneName, LoadSceneMode.Additive);
896:                     else
897:                     {
898:                         Debug.LogWarning($"Scene {newSceneName} is already loaded");
899: 
900:                         // Reset the flag that we disabled before entering this switch
901:                         NetworkClient.isLoadingScene = false;
902:                     }
903:                     break;
904:                 case SceneOperation.UnloadAdditive:
905:                     // Ensure additive scene is actually loaded on client by name or path
906:                     // since we don't know which was passed in the Scene message
907:                     if (SceneManager.GetSceneByName(newSceneName).IsValid() || SceneManager.GetSceneByPath(newSceneName).IsValid())
908:                         loadingSceneAsync = SceneManager.UnloadSceneAsync(newSceneName, UnloadSceneOptions.UnloadAllEmbeddedSceneObjects);
909:                     else
910:                     {
911:                         Debug.LogWarning($"Cannot unload {newSceneName} with UnloadAdditive operation");
912: 
913:                         // Reset the flag that we disabled before entering this switch
914:                         NetworkClient.isLoadingScene = false;
915:                     }
916:                     break;
917:             }
918: 
919:             // don't change the client's current networkSceneName when loading additive scene content
920:             if (sceneOperation == SceneOperation.Normal)
921:                 networkSceneName = newSceneName;
922:         }

974:         {
975:             // NOTE: this cannot use NetworkClient.allClients[0] - that client may be for a completely different purpose.

978:             //Debug.Log("FinishLoadScene: resuming handlers after scene was loading.");
979:             NetworkServer.isLoadingScene = false;
980:             NetworkClient.isLoadingScene = false;

984:             {
985:                 FinishLoadSceneHost();
986:             }
987:             // server-only mode?
988:             else if (mode == NetworkManagerMode.ServerOnly)
989:             {

1004:         void FinishLoadSceneHost()
1005:         {
1006:             // debug message is very important. if we ever break anything then
1007:             // it's very obvious to notice.
1008:             //Debug.Log("Finished loading scene in host mode.");

1011:             {
1012:                 clientLoadedScene = true;
1013:                 clientReadyConnection = null;
1014:             }
1015: 
1016:             // do we need to finish a StartHost() call?
1017:             // then call FinishStartHost and let it take care of spawning etc.
1018:             if (finishStartHostPending)
1019:             {
1020:                 finishStartHostPending = false;
1021:                 FinishStartHost();

1027:                 // the scene change happened because StartHost loaded the

1032:                 // -> once for client OnConnected
1033:                 // -> once in OnClientSceneChanged
1034:             }
1035:             // otherwise we just changed a scene in host mode
1036:             else
1037:             {
1038:                 // spawn server objects
1039:                 NetworkServer.SpawnObjects();

1042:                 OnServerSceneChanged(networkSceneName);
